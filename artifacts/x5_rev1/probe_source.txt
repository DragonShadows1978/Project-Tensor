example : ¬ spRefined (![1, 0] : Fin 2 → ℝ) 0 1 := sp_ex_skip1_of_one where
  sp_ex_skip1_of_one : ¬ spRefined (![1, 0] : Fin 2 → ℝ) 0 1 := by
    unfold spRefined
    rw [not_le]
    have h := le_prefixMax (![1, 0] : Fin 2 → ℝ) (show (0 : Fin 2) ≤ 1 by decide)
    norm_num at h ⊢
    linarith
-- S1 has an active skip; k=j is permitted. A one-key negative-delta trap is avoidable.
example : (1 : ℝ) < prefixMax (![2, 1, 3] : Fin 3 → ℝ) 2 - 0 := by
  have h := le_prefixMax (![2, 1, 3] : Fin 3 → ℝ) (show (2 : Fin 3) ≤ 2 by decide)
  norm_num at h ⊢
  linarith
-- S2: its actual nv type guarantees the opposite of the implication antecedent.
example : sp_mask_superset_nv_stmt →
    ∃ (b : Fin 2 → ℝ) (δ : ℝ) (j : Fin 2), ¬ finalRefined b δ j := by
  rintro ⟨b, δ, j, _, h⟩
  exact ⟨b, δ, j, h⟩
-- S2 positive-antecedent replacement, independent of its theorem.
example : finalRefined (![0, 1] : Fin 2 → ℝ) 0 1 := by
  unfold finalRefined rowMax
  rw [sub_zero]
  apply Finset.sup'_le
  intro i _
  fin_cases i <;> norm_num
-- S3: nonsingleton partition {1,2}; skipped key 1 globally is selected locally.
example : localPrefixMax ({1, 2} : Finset (Fin 3)) (![2, 0, 1] : Fin 3 → ℝ) 1 (by simp) = 0 := by
  unfold localPrefixMax
  apply le_antisymm
  · apply Finset.sup'_le
    intro i hi
    have hip : i ∈ ({1, 2} : Finset (Fin 3)) ∧ i ≤ 1 := Finset.mem_filter.mp hi
    have hi1 : i = 1 := by
      fin_cases i
      · norm_num at hip
      · rfl
      · norm_num at hip
    subst i
    norm_num
  · have h := Finset.le_sup' (![2, 0, 1] : Fin 3 → ℝ)
        (show (1 : Fin 3) ∈ ({1, 2} : Finset (Fin 3)).filter (fun i => i ≤ 1) by decide)
    exact h
-- S4: requested exact endpoints, on actual zRefined, not a numerical emulator.
example : zRefined (![1, 0, 0, 0] : Fin 4 → ℝ) 0 0 ∧

  ⟨(z_not_single_pass 0 le_rfl (Real.sqrt_nonneg 3)).1,

-- Single-key row, delta=0: both masks refine; no skipped-key hypothesis.
example : spRefined (![5] : Fin 1 → ℝ) 0 0 ∧ finalRefined (![5] : Fin 1 → ℝ) 0 0 := by
  constructor <;> norm_num [spRefined, finalRefined, prefixMax, rowMax]
-- S5: errq=0 with an actual skip, epsilon=1, delta=log 1=0 (allowed boundary).

    unfold spRefined

    norm_num at h ⊢
