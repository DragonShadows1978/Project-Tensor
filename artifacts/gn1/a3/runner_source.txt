50: def window(summary, window_file, missing_end):
51:     # Taken: fail-closed parsing and the state table in System-Claude's GW1
52:     # GPU_WINDOW.md (2026). Ours: strict parsing and same-inode mtime validation.
53:     last = None
54:     try:
55:         with open(summary, encoding="utf-8") as stream:
56:             for line in stream:
57:                 match = EVENT.fullmatch(line.rstrip("\n"))
58:                 if match:
59:                     try:
60:                         last = (timestamp(match[1]), match[2])
61:                     except ValueError:
62:                         return Window("busy")
63:     except FileNotFoundError:
64:         return Window("no run", missing_end)
65:     except (OSError, UnicodeError):
66:         return Window("busy")
67:     if last is None:
68:         return Window("busy")
69:     when, event = last
70:     cooldown = COOLDOWN.fullmatch(event)
71:     if cooldown:
72:         try:
73:             end = when + int(cooldown[1])
74:             iso(end)  # Reject unrepresentable timestamps before using them.
75:             return Window("cooldown", end)
76:         except (ValueError, OverflowError, OSError):
77:             return Window("busy")
78:     if not HALT.match(event):
79:         return Window("busy")
80:     try:
81:         with open(window_file, encoding="utf-8") as stream:
82:             contents = stream.read()
83:             modified = os.fstat(stream.fileno()).st_mtime
84:         match = re.fullmatch(r"open-until ([^\r\n]+)\n?", contents)
85:         if match and modified >= when:
86:             end = timestamp(match[1])
87:             if end > time.time():
88:                 return Window("halted", end)
89:     except (OSError, ValueError, OverflowError, UnicodeError):
90:         pass
91:     return Window("halted")

140: def log_line(path, label, state, end, budget, waited, outcome, rc):
141:     # Taken: O_APPEND single-write records (POSIX/IEEE, 1988, year unverified).
142:     # Ours: GW1's nine-field receipt, escaping control separators in user text.
143:     def clean(text):
144:         return str(text).replace("\\", "\\\\").replace("\t", "\\t").replace("\r", "\\r").replace("\n", "\\n")
145: 
146:     try:
147:         fields = (iso(time.time()), os.getcwd(), label, state,
148:                   iso(end) if end is not None else "-", f"{budget:.6f}",
149:                   f"{waited:.6f}", outcome, str(rc))
150:         data = ("\t".join(clean(field) for field in fields) + "\n").encode()
151:         fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_APPEND | os.O_CLOEXEC, 0o666)
152:         try:
153:             if os.write(fd, data) != len(data):
154:                 raise OSError("short log write")
155:         finally:
156:             os.close(fd)
157:     except (OSError, ValueError) as exc:
158:         print(f"forge-gpu-run: warning: cannot write log: {exc}", file=sys.stderr)

161: def supervise(command, deadline, grace, poll, read_deadline):
162:     # Taken: POSIX sessions/process groups (IEEE, 1988) and GNU coreutils
163:     # timeout's TERM-then-KILL supervision (year unverified — lead to check:
164:     # 'POSIX setsid kill process group', 'GNU timeout kill-after history').
165:     # Ours: retain the group through the full grace even when its leader exits.
166:     requested = 0
167: 
168:     def caught(signum, _frame):
169:         nonlocal requested
170:         if not requested:
171:             requested = signum
172: 
173:     old = {sig: signal.signal(sig, caught) for sig in (signal.SIGTERM, signal.SIGINT, signal.SIGHUP)}
174:     try:
175:         try:
176:             child = subprocess.Popen(command, start_new_session=True, close_fds=True)
177:         except OSError as exc:
178:             print(f"forge-gpu-run: cannot start command: {exc}", file=sys.stderr)
179:             return (127 if exc.errno == errno.ENOENT else 126), "ran"
180:         next_poll = time.monotonic() + poll
181:         window_closed = False
182:         while True:
183:             if requested or time.monotonic() >= deadline:
184:                 rc = 128 + requested if requested else 124
185:                 outcome = "ran" if requested else "window-closed" if window_closed else "timeout"
186:                 break
187:             rc = child.poll()
188:             if rc is not None:
189:                 return (rc if rc >= 0 else 128 - rc), "ran"
190:             # TR GW2 step 4a: re-read live ownership without delaying reaping.
191:             # Taken: System-Claude GPU_WINDOW.md (2026), fail-closed policy;
192:             # PEP 418 (2012), monotonic deadlines (unverified — lead to check:
193:             # 'PEP 418 monotonic timeout'). Ours: clamp-only integration with
194:             # GW1's 20ms child checks and existing TERM/grace/group-KILL path.
195:             if time.monotonic() >= next_poll:
196:                 candidate = read_deadline()
197:                 now = time.monotonic()
198:                 if candidate < deadline:
199:                     deadline = candidate
200:                     window_closed = True
201:                 next_poll = now + poll
202:                 if now >= deadline:
203:                     continue
204:             time.sleep(min(0.02, max(0, deadline - time.monotonic())))
205:         try:
206:             os.killpg(child.pid, signal.SIGTERM)
207:         except ProcessLookupError:
208:             pass
209:         grace_end = time.monotonic() + grace
210:         while time.monotonic() < grace_end:
211:             child.poll()
212:             time.sleep(min(0.02, max(0, grace_end - time.monotonic())))
213:         try:
214:             os.killpg(child.pid, signal.SIGKILL)
215:         except ProcessLookupError:
216:             pass
217:         child.wait()
218:         return (128 + requested if requested else rc), outcome
219:     finally:
220:         for sig, handler in old.items():
221:             signal.signal(sig, handler)
