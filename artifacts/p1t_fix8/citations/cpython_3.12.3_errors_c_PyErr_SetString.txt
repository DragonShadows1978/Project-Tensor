# _PyErr_SetString: when the message str cannot be created (PyUnicode_FromString returns NULL), the MemoryError the allocator set stays the raised exception.


void
_PyErr_SetString(PyThreadState *tstate, PyObject *exception,
 const char *string)
{
 PyObject *value = PyUnicode_FromString(string);
 if (value != NULL) {
 _PyErr_SetObject(tstate, exception, value);
 Py_DECREF(value);
 }
}

void
PyErr_SetString(PyObject *exception, const char *string)
{
 PyThreadState *tstate = _PyThreadState_GET();
 _PyErr_SetString(tstate, exception, string);
}
